cmake_minimum_required(VERSION 3.20)
# Every dependency must use the same architecture as the plugin.
if(CMAKE_HOST_APPLE OR CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "macOS architecture (Apple Silicon only)")
  if(NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    message(FATAL_ERROR "macOS builds support Apple Silicon only; use -DCMAKE_OSX_ARCHITECTURES=arm64")
  endif()
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
  set(CMAKE_C_VISIBILITY_PRESET hidden)
  set(CMAKE_CXX_VISIBILITY_PRESET hidden)
  set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
endif()
project(ToLissOANS VERSION 0.4.37 LANGUAGES C CXX)

  set(OANS_PACKAGE_NAME "ToLissOANS-debug")
endif()

set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
  set(SDK_ROOT "E:/cxx_project/_lib/SDK" CACHE PATH "X-Plane SDK root")
else()
  set(SDK_ROOT "$ENV{XPLANE_SDK_ROOT}" CACHE PATH "X-Plane SDK root (4.0 or newer for arm64)")
endif()
add_library(oans_core STATIC src/core.cpp src/fms.cpp src/runway_ahead.cpp)
set(LZMA_DIR "${CMAKE_SOURCE_DIR}/vendor/xptools/lzma19/C")
set(LZMA_NAMES 7zAlloc 7zArcIn 7zBuf 7zBuf2 7zCrc 7zCrcOpt 7zDec 7zFile 7zStream Bcj2 Bra Bra86 BraIA64 CpuArch Delta Lzma2Dec LzmaDec)
set(LZMA_SOURCES)
foreach(name IN LISTS LZMA_NAMES)
  list(APPEND LZMA_SOURCES "${LZMA_DIR}/${name}.c")
endforeach()
add_library(oans_md5 STATIC vendor/xptools/Utils/md5.c)
target_include_directories(oans_md5 PUBLIC vendor/xptools/Utils)
add_library(oans_dsf_reader STATIC vendor/xptools/DSF/DSFLib.cpp vendor/xptools/Utils/XChunkyFileUtils.cpp ${LZMA_SOURCES})
target_link_libraries(oans_dsf_reader PUBLIC oans_md5)
target_include_directories(oans_dsf_reader PUBLIC vendor/xptools/DSF vendor/xptools/Utils "${LZMA_DIR}")
target_compile_definitions(oans_dsf_reader PUBLIC LIL=1 NOMINMAX PRIVATE BIG=0 USE_7Z=0)
add_library(oans_dsf STATIC src/dsf_buildings.cpp)
target_link_libraries(oans_dsf PUBLIC oans_core PRIVATE oans_dsf_reader)
add_executable(dsf_buildings_tests tests/dsf_buildings_tests.cpp)
target_link_libraries(dsf_buildings_tests PRIVATE oans_dsf)
add_executable(dsf_buildings_probe tools/dsf_buildings_probe.cpp)
target_link_libraries(dsf_buildings_probe PRIVATE oans_dsf)
# Explicit paths also work in the exported code[develop purpose only] directory;
# its brackets must not be interpreted as a glob character class.
set(LUA_NAMES lapi lauxlib lbaselib lcode lcorolib lctype ldblib ldebug ldo ldump lfunc lgc linit liolib llex lmathlib lmem loadlib lobject lopcodes loslib lparser lstate lstring lstrlib ltable ltablib ltm lundump lutf8lib lvm lzio)

add_library(oans_lua STATIC ${LUA_SOURCES})

target_link_libraries(a350_map PUBLIC oans_core PRIVATE oans_lua)
add_executable(lua_vertex_buffer_tests tests/lua_vertex_buffer_tests.cpp)
target_link_libraries(lua_vertex_buffer_tests PRIVATE oans_lua)
target_include_directories(lua_vertex_buffer_tests PRIVATE src)
add_executable(map_vertex_probe tools/map_vertex_probe.cpp)
target_link_libraries(map_vertex_probe PRIVATE a350_map)
add_executable(map_reuse_probe tools/map_reuse_probe.cpp)
target_link_libraries(map_reuse_probe PRIVATE a350_map)
target_include_directories(oans_core PUBLIC src)
add_executable(oans_tests tests/core_tests.cpp)
target_link_libraries(oans_tests PRIVATE oans_core)
add_executable(apt_inspect tools/apt_inspect.cpp)
add_executable(a350_map_probe tools/a350_map_probe.cpp)
target_link_libraries(a350_map_probe PRIVATE a350_map)
target_link_libraries(apt_inspect PRIVATE oans_core)

  set_target_properties(ToLissOANS PROPERTIES PREFIX "" SUFFIX ".xpl" OUTPUT_NAME "win"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/dist/${OANS_PACKAGE_NAME}/64")
  if(OANS_DEBUG)
    add_executable(CollectLogs tools/collect_logs.cpp)
    set_target_properties(CollectLogs PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/dist/${OANS_PACKAGE_NAME}")

    target_link_options(fms_tests PRIVATE -static)

if(APPLE)
  set(CMAKE_OBJCXX_VISIBILITY_PRESET hidden)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  if(NOT EXISTS "${SDK_ROOT}/CHeaders/XPLM/XPLMPlugin.h")
    message(FATAL_ERROR "Set SDK_ROOT to an X-Plane SDK 4.0+ directory")
  endif()
  find_library(OANS_XPLM XPLM PATHS "${SDK_ROOT}/Libraries/Mac" NO_DEFAULT_PATH REQUIRED)
  find_library(OANS_OPENGL OpenGL REQUIRED)
  find_library(OANS_APPKIT AppKit REQUIRED)
  # Use macOS's libcurl, avoiding Homebrew runtime dependencies in the package.
  execute_process(COMMAND xcrun --sdk macosx --show-sdk-path
    OUTPUT_VARIABLE OANS_MACOS_SDK OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  find_library(OANS_CURL curl PATHS "${CMAKE_OSX_SYSROOT}/usr/lib" "${OANS_MACOS_SDK}/usr/lib" NO_DEFAULT_PATH REQUIRED)
  add_library(ToLissOANS SHARED src/plugin.cpp src/aircraft_profile_generator.cpp src/nd_ui.cpp src/fms_xplm.cpp src/mac_platform.mm)
  target_include_directories(ToLissOANS PRIVATE "${SDK_ROOT}/CHeaders/XPLM")
  target_compile_definitions(ToLissOANS PRIVATE IBM=0 APL=1 LIN=0 XPLM200 XPLM210 XPLM300 XPLM301 XPLM303 GL_SILENCE_DEPRECATION)
  target_compile_options(ToLissOANS PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_libraries(ToLissOANS PRIVATE oans_core a350_map oans_dsf oans_md5 "${OANS_XPLM}" "${OANS_OPENGL}" "${OANS_APPKIT}" "${OANS_CURL}")
  # X-Plane supplies the real XPLM framework; never embed the SDK's stub or its path.
  set_target_properties(ToLissOANS PROPERTIES PREFIX "" SUFFIX ".xpl" OUTPUT_NAME "mac"
    SKIP_BUILD_RPATH TRUE
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/dist/${OANS_PACKAGE_NAME}/64")
  add_executable(gl_state_tests tests/gl_state_tests.cpp)
  target_link_libraries(gl_state_tests PRIVATE "${OANS_OPENGL}")
  add_test(NAME oans_gl_state COMMAND gl_state_tests)
  set_tests_properties(oans_gl_state PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_executable(detail_log_tests tests/detail_log_tests.cpp)
  target_include_directories(detail_log_tests PRIVATE src)
  add_test(NAME detail_log COMMAND detail_log_tests "${CMAKE_BINARY_DIR}/detail-log-tests")
  install(TARGETS ToLissOANS LIBRARY DESTINATION 00_${OANS_PACKAGE_NAME}/64)
  install(DIRECTORY assets cache config DESTINATION 00_${OANS_PACKAGE_NAME})
  install(FILES config.ini LICENSE THIRD_PARTY_NOTICES.txt README.md DESTINATION 00_${OANS_PACKAGE_NAME})
  install(FILES docs/MACOS.md DESTINATION 00_${OANS_PACKAGE_NAME})
endif()
